#pragma config OSC=HS
#pragma config WDT=OFF
#include<xc.h>

void main()
{
    TRISD=0;
    LATD=0x01;
    T0CON=0x07;
    
    while(1)
    {
        TMR0H=0xB3;
        TMR0L=0xB4;
        T0CONbits.TMR0ON=1;
        while(INTCONbits.TMR0IF==0);
        T0CONbits.TMR0ON=0;
        INTCONbits.TMR0IF=0;
        LATD=(LATD<<1)|(LATD>>7);
    }
}
